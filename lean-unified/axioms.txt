UnifiedEJM.unified_ejm_family {n : ℕ} (hn : 5 ≤ n) :
  (∀ (t : ℝ) (k l : OddEJM.Bits n),
      OddEJM.inner (UnifiedEJM.measurement n t k) (UnifiedEJM.measurement n t l) = if k = l then 1 else 0) ∧
    (∀ (t : ℝ) (label : OddEJM.Bits n) (j : Fin n),
        have v :=
          OddEJM.Geometry.vertex (OddEJM.localOrientation j) (OddEJM.localOutcome j label).1
            (OddEJM.localOutcome j label).2;
        OddEJM.reduced (UnifiedEJM.measurement n t label) j =
          (1 / 2) •
            (1 + ↑(OddEJM.radius n t) • (↑(v 0) • OddEJM.sigmaX + ↑(v 1) • OddEJM.sigmaY + ↑(v 2) • OddEJM.sigmaZ))) ∧
      (∀ (j : Fin n) (q s : OddEJM.Bit),
          OddEJM.Geometry.dot3 (OddEJM.Geometry.vertex (OddEJM.localOrientation j) q s)
              (OddEJM.Geometry.vertex (OddEJM.localOrientation j) q s) =
            1) ∧
        (∀ (j : Fin n) (a b : OddEJM.Bit × OddEJM.Bit),
            a ≠ b →
              OddEJM.Geometry.dot3 (OddEJM.Geometry.vertex (OddEJM.localOrientation j) a.1 a.2)
                  (OddEJM.Geometry.vertex (OddEJM.localOrientation j) b.1 b.2) =
                -(1 / 3)) ∧
          (∀ (j : Fin n) (a : OddEJM.Bit × OddEJM.Bit),
              Nat.card { label // OddEJM.localOutcome j label = a } = 2 ^ (n - 2)) ∧
            OddEJM.radius n '' OddEJM.Geometry.parameterInterval = Set.Icc 0 (OddEJM.Geometry.radiusMax n) ∧
              Continuous (OddEJM.radius n) ∧
                StrictAntiOn (OddEJM.radius n) OddEJM.Geometry.parameterInterval ∧
                  (∀ (label : OddEJM.Bits n),
                      UnifiedEJM.measurement n 0 label =
                        OddEJM.circuit fun z =>
                          OddEJM.normFactor n * OddEJM.originalDiagonal z * OddEJM.character label z) ∧

                        UnifiedEJM.measurement n t label =
                          OddEJM.circuit
                            (OddEJM.phaseGate (UnifiedEJM.rawPhase (OddEJM.anglePhase t))
                              (OddEJM.tensorHadamard (OddEJM.ket label)))) ∧
                      ∀ (t : ℝ) (z w : OddEJM.Bits n),
                        ∑ label, UnifiedEJM.measurement n t label z * star (UnifiedEJM.measurement n t label w) =
                          if z = w then 1 else 0

              UnifiedEJM.Regularity.bloch (UnifiedEJM.Physical.measurement u label) j = v (a label)) ∧
